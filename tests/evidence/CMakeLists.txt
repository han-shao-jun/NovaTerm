cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NovaTermEvidenceTests LANGUAGES CXX)
    set(CMAKE_AUTOMOC ON)
    include(CTest)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Test)
    add_subdirectory(../../src/evidence evidence)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
qt_add_executable(novaterm_evidence_tests EvidenceTests.cpp)
target_link_libraries(novaterm_evidence_tests PRIVATE novaterm_evidence Qt6::Test)
set_target_properties(novaterm_evidence_tests PROPERTIES WIN32_EXECUTABLE FALSE)
add_test(NAME novaterm_evidence_tests COMMAND novaterm_evidence_tests
    -o "${CMAKE_CURRENT_BINARY_DIR}/EvidenceTests.ctest.txt,txt")
set_tests_properties(novaterm_evidence_tests PROPERTIES LABELS "unit;evidence;history;p8" TIMEOUT 30)
