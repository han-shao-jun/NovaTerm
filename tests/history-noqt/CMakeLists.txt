cmake_minimum_required(VERSION 3.20)
project(NovaTermHistoryCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 复用主工程的依赖输出规范化，避免中文 /showIncludes 导致 Ninja 漏依赖。
if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
    find_program(_novaterm_powershell NAMES powershell REQUIRED)
    get_filename_component(_history_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    file(SHA256 "${_history_root}/scripts/msvc-deps-launcher.cmake.in" _novaterm_launcher_hash)
    file(SHA256 "${_history_root}/scripts/normalize-msvc-output.ps1" _novaterm_normalizer_hash)
    set(_history_project_source "${PROJECT_SOURCE_DIR}")
    set(PROJECT_SOURCE_DIR "${_history_root}")
    configure_file("${_history_root}/scripts/msvc-deps-launcher.cmake.in"
        "${CMAKE_BINARY_DIR}/msvc-deps-launcher.cmake" @ONLY)
    set(PROJECT_SOURCE_DIR "${_history_project_source}")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
    set(CMAKE_C_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "Note: including file: ")
    set(CMAKE_C_COMPILER_LAUNCHER ${CMAKE_COMMAND} -P "${CMAKE_BINARY_DIR}/msvc-deps-launcher.cmake" --)
    set(CMAKE_CXX_COMPILER_LAUNCHER ${CMAKE_COMMAND} -P "${CMAKE_BINARY_DIR}/msvc-deps-launcher.cmake" --)
endif()
include(CTest)
add_subdirectory(../../src/history history)
add_subdirectory(../history checks)
